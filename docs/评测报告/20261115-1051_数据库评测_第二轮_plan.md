<details><summary>S01：各品类的销售额分别是多少？</summary>

```sql
SELECT
  p.category AS 品类,
  ROUND(SUM(oi.unit_price * oi.quantity), 2) AS 销售额
FROM order_items oi
JOIN products p ON oi.product_id = p.product_id
GROUP BY p.category
ORDER BY 销售额 DESC
```

</details>

<details><summary>S02：钓竿的退货率是多少？</summary>

```sql
SELECT
    p.category AS 品类,
    COUNT(DISTINCT oi.order_id) AS 订单数,
    COUNT(DISTINCT CASE WHEN oi.is_returned = 1 THEN oi.order_id END) AS 退货订单数,
    ROUND(COUNT(DISTINCT CASE WHEN oi.is_returned = 1 THEN oi.order_id END) * 1.0
          / COUNT(DISTINCT oi.order_id), 4) AS 退货率
FROM order_items oi
JOIN products p ON oi.product_id = p.product_id
WHERE p.category = '钓竿'
GROUP BY p.category
```

</details>

<details><summary>S03：各店铺分别有多少个订单、销售额是多少？</summary>

```sql
SELECT
  s.shop_name AS 店铺名称,
  COUNT(DISTINCT o.order_id) AS 订单数,
  ROUND(SUM(oi.unit_price * oi.quantity), 2) AS 销售额
FROM orders o
JOIN shops s ON o.shop_id = s.shop_id
JOIN order_items oi ON o.order_id = oi.order_id
GROUP BY s.shop_id, s.shop_name
ORDER BY 销售额 DESC
```

</details>

<details><summary>S04：哪个地区的客单价最高？是多少？</summary>

```sql
WITH region_stats AS (
  SELECT
    o.region AS 地区,
    ROUND(SUM(oi.unit_price * oi.quantity), 2) AS 销售额,
    COUNT(DISTINCT o.order_id) AS 订单数
  FROM orders o
  JOIN order_items oi ON oi.order_id = o.order_id
  GROUP BY o.region
)
SELECT
  地区,
  销售额,
  订单数,
  ROUND(销售额 * 1.0 / 订单数, 4) AS 客单价
FROM region_stats
ORDER BY 客单价 DESC
LIMIT 1
```

</details>

<details><summary>S05：2025 年哪个月的销售额最高？是多少？</summary>

```sql
WITH monthly_sales AS (
  SELECT
    strftime('%Y-%m', o.order_date) AS 月份,
    ROUND(SUM(oi.unit_price * oi.quantity), 2) AS 销售额
  FROM orders o
  JOIN order_items oi ON oi.order_id = o.order_id
  WHERE strftime('%Y', o.order_date) = '2025'

LIMIT 1
```

<details><summary>S06：各店铺的广告投入产出比（ROAS）是多少？哪个最高？</summary>

```sql
WITH sales AS (
    SELECT o.shop_id AS shop_id,
           SUM(oi.unit_price * oi.quantity) AS 销售额
    FROM order_items oi
    JOIN orders o ON oi.order_id = o.order_id
    GROUP BY o.shop_id
),
ads AS (
    SELECT shop_id AS shop_id,
           SUM(spend) AS 广告花费
    FROM ad_spend
    GROUP BY shop_id
)
SELECT s.shop_name AS 店铺,
       ROUND(sa.销售额, 2) AS 销售额,
       ROUND(a.广告花费, 2) AS 广告花费,
       ROUND(sa.销售额 / a.广告花费, 4) AS ROAS,
       RANK() OVER (ORDER BY sa.销售额 / a.广告花费 DESC) AS ROAS排名
FROM sales sa
JOIN ads a ON sa.shop_id = a.shop_id
JOIN shops s ON sa.shop_id = s.shop_id
ORDER BY ROAS DESC
```

</details>

<details><summary>S11：各品类的 ACOS（广告花费占销售额的比例）是多少？</summary>

```sql
WITH spend AS (
    SELECT category AS 品类,
           SUM(spend) AS 广告花费
    FROM ad_spend
    GROUP BY category
),
sales AS (
    SELECT p.category AS 品类,
           SUM(oi.unit_price * oi.quantity) AS 销售额
    FROM order_items oi
    JOIN products p ON oi.product_id = p.product_id
    GROUP BY p.category
)
SELECT COALESCE(s.品类, sa.品类) AS 品类,
       ROUND(COALESCE(s.广告花费, 0), 2) AS 广告花费,
       ROUND(COALESCE(sa.销售额, 0), 2) AS 销售额,
       ROUND(COALESCE(s.广告花费, 0) / NULLIF(sa.销售额, 0), 4) AS ACOS
FROM spend s
LEFT JOIN sales sa ON s.品类 = sa.品类
UNION
SELECT COALESCE(s.品类, sa.品类) AS 品类,
       ROUND(COALESCE(s.广告花费, 0), 2) AS 广告花费,
       ROUND(COALESCE(sa.销售额, 0), 2) AS 销售额,
       ROUND(COALESCE(s.广告花费, 0) / NULLIF(sa.销售额, 0), 4) AS ACOS
FROM sales sa
LEFT JOIN spend s ON s.品类 = sa.品类
WHERE s.品类 IS NULL
ORDER BY 品类
```
